  if (ch === 1 && t >= 3.2) find("~/code/orbit").s = t < 6.4 ? "working" : "done";

{/* Mintlify 文档说明 MDX 支持 useState 和 useEffect，
   因此可变时钟和元素句柄保存在 state 中，而不是 ref 中。

   此文件每个 export 的函数体内都不要放置 JS 注释，包括行注释和块注释：
   托管的 Mintlify 构建无法解析含有此类注释的页面，
   只会显示 "parsing error" 占位内容。mint dev 和 mint validate 都会接受它，
   因此问题只会在线上站点出现。请像这样在顶层添加注释。 */}
export const Demo = () => {
  const [st, setSt] = useState({ ch: 0, t: 0 });
  const [c0] = useState(() => ({ ch: 0, start: 0, visible: true, last: 0, paused: 0 }));
  const [el, setEl] = useState(null);
  useEffect(() => {
    const c = c0;
    c.start = performance.now();
    let raf = 0;
    const tick = (now) => {
      raf = requestAnimationFrame(tick);
      if (!c.visible || document.hidden) { c.start = now - c.paused; return; }
      let t = (now - c.start) / 1000;
      if (t >= CHAPTERS[c.ch].d) { c.ch = (c.ch + 1) % CHAPTERS.length; c.start = now; t = 0; }
      c.paused = now - c.start;
      if (now - c.last < 33) return;
      c.last = now;
      setSt({ ch: c.ch, t });
    };
    c.paused = 0;
    raf = requestAnimationFrame(tick);
    const io = new IntersectionObserver(([e]) => { c.visible = e.isIntersecting; }, { threshold: 0.05 });
    if (el) io.observe(el);
    return () => { cancelAnimationFrame(raf); io.disconnect(); };
  }, [el]);
  const go = (i) => {
    const c = c0;
    c.ch = i; c.start = performance.now(); c.paused = 0;
    setSt({ ch: i, t: 0 });
  };
  const cur = CHAPTERS[st.ch];
  return (
    <div className="lyrel-home-demo" ref={setEl}>
      <Frame ch={st.ch} t={st.t} />
      <div className="lyrel-home-chapters" role="tablist" aria-label="lyrel 演示">
        {CHAPTERS.map((c, i) => (
          <button key={c.key} role="tab" aria-selected={i === st.ch} className={"lyrel-home-chapter" + (i === st.ch ? " is-on" : "")} onClick={() => go(i)}>
            <span className="lyrel-home-chapter-bar"><span style={{ width: (i === st.ch ? Math.min(1, st.t / cur.d) * 100 : i < st.ch ? 100 : 0) + "%" }} /></span>
            <span className="lyrel-home-chapter-label">{c.label}</span>
          </button>
        ))}
      </div>
      <div className="lyrel-home-chapter-now">{cur.label}</div>
    </div>
  );
};
